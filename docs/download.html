Quick Start (Python)
import pandas as pd # Load 16S genus table (primary pretraining feature matrix) X_16s = pd.read_parquet("data/16s/corpus_16s_genus_relab_expanded.parquet") \ .set_index("sample_id") # 164,958 × 977 # Load shotgun species table X_shg = pd.read_parquet("data/shotgun/corpus_shotgun_species_relab.parquet") \ .set_index("sample_uid") # 22,583 × 713 # Load master metadata meta = pd.read_parquet("data/metadata/corpus_metadata_catalog_expanded.parquet") \ .set_index("sample_uid") # 187,546 × 15 # Always split by modality before downstream analysis! meta_16s = meta[meta.modality == "16S"] meta_shg = meta[meta.modality == "shotgun"]
